ADMIN_BIT = 0b100   # permission: admin (bit 2)
GRADER_BIT = 0b010  # permission: grader (bit 1)
VIEWER_BIT = 0b001  # permission: viewer (bit 0)
PASS_MARK = 40
 
 
def review_submissions(submissions, staff_flags):
    """
    submissions: list of dicts - each has 'student_id' (str), 'score' (int),
                 'minutes' (int)
    staff_flags: int, bit-encoded permissions of the staff member
    Returns    : set of student IDs flagged for manual review
    """
    if not submissions:
        print("No submissions to review")
        return set()
 
    needed = ADMIN_BIT | GRADER_BIT
    if (staff_flags & needed) == needed:
        print("Full report access granted")
    else:
        print("Limited access - summary only")
 
    flagged = set()
    unique_students = set()
    total_minutes = 0
    scores = []
 
    for s in submissions:
        sid = s["student_id"].strip().upper()
        unique_students.add(sid)
        scores.append(s["score"])
        total_minutes += s["minutes"]
 
        if s["score"] < PASS_MARK:
            flagged.add(sid)
 
        if s["score"] == 100 and s["minutes"] < 5:
            flagged.add(sid)
 
    print("Unique students:", len(unique_students))
 
    hours, mins = total_minutes // 60, total_minutes % 60
    print(f"Total exam time: {hours} h {mins} min")
 
    ranked = sorted(scores, reverse=True)
    top3 = ranked[:3]
    print("Top 3 scores:", top3)
 
    print("Flagged students:", sorted(flagged))
    return flagged

# ---------------- Input / output (do not change) ----------------
import json
data = json.loads(input())
result = review_submissions(data["submissions"], data["staff_flags"])
print("Returned:", sorted(result))
Q2

Correct
The FitPulse Check-in Audit
Your Answer

python
MANAGER_BIT = 0b100     # permission: manager (bit 2)
TRAINER_BIT = 0b010     # permission: trainer (bit 1)
FRONT_DESK_BIT = 0b001  # permission: front desk (bit 0)
MAX_MINUTES = 120
 
 
def audit_checkins(checkins, staff_flags):
    """
    checkins   : list of dicts - each has 'member_id' (str), 'minutes' (int),
                 'fee' (float), 'guest' (bool)
    staff_flags: int, bit-encoded permissions of the staff member
    Returns    : set of member IDs flagged for manual review
    """
    if not checkins:
        print("No check-ins to audit")
        return set()
 
    # 1. Full report is only for staff who are BOTH manager and trainer
    needed = MANAGER_BIT | TRAINER_BIT
    if (staff_flags & needed) == needed:
        print("Full report access granted")
    else:
        print("Limited access - summary only")
 
    flagged = set()
 
    # 2. Flag sessions of 120 minutes or more
    for c in checkins:
        if c["minutes"] >= MAX_MINUTES:
            flagged.add(c["member_id"])
 
    # 3. Flag invalid records: negative minutes OR negative fee
    for c in checkins:
        if c["minutes"] < 0 or c["fee"] < 0:
            flagged.add(c["member_id"])
 
    # 4. Flag free check-ins made by members (not guests)
    for c in checkins:
        if c["fee"] == 0 and not c["guest"]:
            flagged.add(c["member_id"])
 
    # 5. Percentage of guest check-ins
    guest_count = sum(1 for c in checkins if c["guest"])
    guest_pct = guest_count / len(checkins) * 100
    print(f"Guest check-ins: {guest_pct:.1f}%")
 
    # 6. Show the most recent check-in (last record in the list)
    last = checkins[-1]
    print("Last check-in:", last["member_id"])
 
    print("Flagged members:", sorted(flagged))
    return flagged
 
 
# ---------------- Input / output (do not change) ----------------
import json
 
data = json.loads(input())
result = audit_checkins(data["checkins"], data["staff_flags"])
print("Returned:", sorted(result))

member, miles = map(int, input().split())

if member == 1:
    if miles >= 100000:
        print("Eligible: Platinum Upgrade")
    elif miles >= 50000:
        print("Eligible: Gold Upgrade")
    else:
        print("Not Eligible: Insufficient Miles")
else:
    if miles >= 100000:
        print("Eligible: Standard Upgrade Offer")
    else:
        print("Not Eligible: Join Membership Program")
Explanation

The outer if / else splits the problem in two completely separate worlds: members and non-members. Once
you're inside one of those branches, you only ever think about that branch's own rules -- a member's
50,000-mile threshold has nothing to do with a non-member's 100,000-mile threshold, so they can't be checked
with a single flat elif chain. Inside the member branch, an elif chain checks miles from the highest tier down (>= 100000 first, then >= 50000), falling through to "Not Eligible" only if neither matches. Inside the non-member branch, the same
top-down idea applies but with just one threshold to check.
